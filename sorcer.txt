-- ═══ HTTP BACKEND CONFIG ══════════════════════════════════
-- npoint.io document URL — shared storage for admin commands and bans
local NPOINT_URL = "https://api.npoint.io/530e53a55a5329367390"
local HttpService = game:GetService("HttpService")

-- Find the HTTP request function (different executors name it differently)
local httpRequest = request or http_request or (syn and syn.request) or (http and http.request)
if httpRequest then
    pcall(function() HttpService.HttpEnabled = true end)
end

-- Fetch the backend JSON (returns table or nil)
local function fetchBackend()
    if not httpRequest then return nil end
    local ok, response = pcall(function()
        return httpRequest({
            Url = NPOINT_URL,
            Method = "GET",
        })
    end)
    if not ok or not response or response.StatusCode ~= 200 then return nil end
    local ok2, data = pcall(function() return HttpService:JSONDecode(response.Body) end)
    if not ok2 or type(data) ~= "table" then return nil end
    -- Ensure commands is an array
    if type(data.commands) ~= "table" then data.commands = {} end
    -- Ensure bans is an object (not array)
    if type(data.bans) ~= "table" then data.bans = {} end
    if data.bans[1] ~= nil then
        -- It's an array — convert to object
        local fixed = {}
        for _, b in ipairs(data.bans) do
            if b and b.userId then fixed[tostring(b.userId)] = b end
        end
        data.bans = fixed
    end
    -- Ensure online is an object (not array)
    if type(data.online) ~= "table" then data.online = {} end
    if data.online[1] ~= nil then
        local fixed = {}
        for _, u in ipairs(data.online) do
            if u and u.userId then fixed[tostring(u.userId)] = u end
        end
        data.online = fixed
    end
    return data
end

-- Save the backend JSON
-- We manually build the JSON to ensure bans/online are always objects {} not arrays []
local function saveBackend(data)
    if not httpRequest then return false end

    -- Build JSON manually for the critical parts
    local parts = {}

    -- bans (always object)
    table.insert(parts, '"bans":{')
    local banParts = {}
    if type(data.bans) == "table" and data.bans[1] == nil then
        for k, v in pairs(data.bans) do
            if k ~= "__dummy" then
                table.insert(banParts, '"' .. k .. '":' .. HttpService:JSONEncode(v))
            end
        end
    end
    table.insert(parts, table.concat(banParts, ","))
    table.insert(parts, '}')

    -- commands (always array)
    table.insert(parts, ',"commands":')
    if type(data.commands) == "table" then
        table.insert(parts, HttpService:JSONEncode(data.commands))
    else
        table.insert(parts, "[]")
    end

    -- online (always object)
    table.insert(parts, ',"online":{')
    local onlineParts = {}
    if type(data.online) == "table" and data.online[1] == nil then
        for k, v in pairs(data.online) do
            if k ~= "__dummy" then
                table.insert(onlineParts, '"' .. k .. '":' .. HttpService:JSONEncode(v))
            end
        end
    end
    table.insert(parts, table.concat(onlineParts, ","))
    table.insert(parts, '}')

    local json = "{" .. table.concat(parts, "") .. "}"

    local ok, response = pcall(function()
        return httpRequest({
            Url = NPOINT_URL,
            Method = "POST",
            Headers = { ["Content-Type"] = "application/json" },
            Body = json,
        })
    end)
    return ok and response and response.StatusCode == 200
end

-- Check if a player is an admin
local function checkIsAdmin(player)
    for _, admin in ipairs(ADMINS) do
        if type(admin) == "number" then
            if player.UserId == admin then return true end
        else
            if player.Name == admin or player.DisplayName == admin then return true end
        end
    end
    return false
end

-- ═══ BAN SYSTEM (npoint-backed, persists cross-server) ════
local frozenPlayers = {}  -- { [userId] = true }
local executedCmdIds = {} -- { [cmdId] = true } — commands we already ran

-- Cache of backend data (refreshed by polling loop)
local backendCache = { commands = {}, bans = {} }

-- Check if a player is banned based on backend data
local function isPlayerBanned(player)
    if checkIsAdmin(player) then return false end
    -- Safety: ensure bans is an object, not an array
    if type(backendCache.bans) ~= "table" then return false end
    if backendCache.bans[1] ~= nil then return false end  -- it's an array, no bans
    local ban = backendCache.bans[tostring(player.UserId)]
    if not ban then return false end
    if ban.expires == nil then return true, nil end  -- infinite
    if os.time() < ban.expires then return true, ban.expires end
    return false  -- expired
end

-- Admin adds a ban to the backend
local function adminAddBan(player, durationSeconds)
    local data = fetchBackend()
    if not data then return false end
    -- Ensure bans is an object, not an array
    if type(data.bans) ~= "table" then data.bans = {} end
    -- Convert array to object if needed
    if data.bans[1] ~= nil then
        local fixed = {}
        for _, b in ipairs(data.bans) do
            if b.userId then fixed[tostring(b.userId)] = b end
        end
        data.bans = fixed
    end
    data.bans[tostring(player.UserId)] = {
        name = player.Name,
        displayName = player.DisplayName,
        expires = durationSeconds and (os.time() + durationSeconds) or nil,
        adminId = LocalPlayer.UserId,
    }
    -- Also post a ban command so the victim gets notified immediately
    local cmdId = tostring(tick() + math.random())
    if type(data.commands) ~= "table" then data.commands = {} end
    table.insert(data.commands, {
        id = cmdId,
        action = "ban",
        victimId = player.UserId,
        adminId = LocalPlayer.UserId,
        timestamp = os.time(),
        extraData = tostring(durationSeconds or "perm"),
    })
    saveBackend(data)
    backendCache = data
    return true
end

-- Admin removes a ban from the backend
local function adminRemoveBan(userIdStr)
    local data = fetchBackend()
    if not data then return false end
    data.bans[userIdStr] = nil
    saveBackend(data)
    backendCache = data
    return true
end

-- ═══ SCRIPT USER DETECTION (npoint-backed) ════════════════
-- Client attributes don't replicate, so we register ourselves on the npoint
-- backend. The admin polls the backend to see who's running the script.
local onlineUsers = {}  -- { [userId_string] = { name = "...", displayName = "...", lastSeen = timestamp } }
local pollingActive = true  -- moved up so register loop can see it

-- Register ourselves as online
local function registerOnline()
    local data = fetchBackend()
    if not data then return end
    if not data.online then data.online = {} end
    data.online[tostring(LocalPlayer.UserId)] = {
        name = LocalPlayer.Name,
        displayName = LocalPlayer.DisplayName,
        lastSeen = os.time(),
    }
    saveBackend(data)
    onlineUsers = data.online
end

-- Check if a player is a script user (based on backend data)
local function isScriptUser(player)
    local entry = onlineUsers[tostring(player.UserId)]
    if not entry then return false end
    -- Consider online if seen in the last 15 seconds
    return (os.time() - entry.lastSeen) < 15
end

-- Register in background (don't block script load)
task.spawn(function()
    registerOnline()
    while pollingActive do
        task.wait(5)
        if banOverlayActive then break end
        registerOnline()
    end
end)

-- Clean up our entry when leaving the game (local script safe)
Players.PlayerRemoving:Connect(function(player)
    if player == LocalPlayer then
        local data = fetchBackend()
        if data and data.online then
            data.online[tostring(LocalPlayer.UserId)] = nil
            saveBackend(data)
        end
    end
end)

-- ═══ VICTIM ACTIONS (executed on self when receiving command) ═══
local function executeVictimAction(action, extraData)
    if action == "kick" then
        pcall(function() LocalPlayer:Kick("Kicked by Sorcery Hub admin") end)
    elseif action == "kill" then
        pcall(function()
            local char = LocalPlayer.Character
            if char then
                local h = char:FindFirstChildOfClass("Humanoid")
                if h then h.Health = 0 end
            end
        end)
    elseif action == "freeze" then
        pcall(function()
            local char = LocalPlayer.Character
            if char then
                local h = char:FindFirstChildOfClass("Humanoid")
                if h then
                    h.WalkSpeed = 0
                    h.JumpPower = 0
                    h.JumpHeight = 0
                end
                for _, part in ipairs(char:GetDescendants()) do
                    if part:IsA("BasePart") then part.Anchored = true end
                end
                frozenPlayers[LocalPlayer.UserId] = true
            end
        end)
    elseif action == "unfreeze" then
        pcall(function()
            local char = LocalPlayer.Character
            if char then
                local h = char:FindFirstChildOfClass("Humanoid")
                if h then
                    h.WalkSpeed = 16
                    h.JumpPower = 50
                    h.JumpHeight = 7.2
                end
                for _, part in ipairs(char:GetDescendants()) do
                    if part:IsA("BasePart") then part.Anchored = false end
                end
                frozenPlayers[LocalPlayer.UserId] = nil
            end
        end)
    elseif action == "bring" then
        pcall(function()
            local adminId = tonumber(extraData)
            if not adminId then return end
            local adminPlayer = Players:GetPlayerByUserId(adminId)
            if not adminPlayer then return end
            local myChar = LocalPlayer.Character
            local adminChar = adminPlayer.Character
            if myChar and adminChar then
                local myRoot = myChar:FindFirstChild("HumanoidRootPart")
                local adminRoot = adminChar:FindFirstChild("HumanoidRootPart")
                if myRoot and adminRoot then
                    myRoot.CFrame = adminRoot.CFrame * CFrame.new(0, 0, -4)
                end
            end
        end)
    elseif action == "ban" then
        -- extraData = "perm" or seconds as string
        local duration = nil
        if extraData and extraData ~= "perm" then
            duration = tonumber(extraData)
        end
        showBanOverlay(duration)
    end
end

-- ═══ COMMAND POLLING LOOP ═════════════════════════════════
-- Every 2 seconds, fetch the backend and check for new commands targeting us
task.spawn(function()
    while pollingActive do
        task.wait(2)
        -- Stop polling if ban overlay is active
        if banOverlayActive then break end

        -- Wrap everything in pcall so a single error doesn't kill the loop
        local ok, err = pcall(function()
            local data = fetchBackend()
            if not data then return end

            -- Ensure data structures are correct types
            if type(data.commands) ~= "table" then data.commands = {} end
            if type(data.bans) ~= "table" then data.bans = {} end
            -- Fix bans if it's an array instead of object
            if data.bans[1] ~= nil then
                local fixed = {}
                for _, b in ipairs(data.bans) do
                    if b and b.userId then fixed[tostring(b.userId)] = b end
                end
                data.bans = fixed
            end

            backendCache = data
            local cmdsToExecute = {}
            local cmdsToRemove = {}

            -- First pass: find commands targeting us
            for _, cmd in ipairs(data.commands) do
                if type(cmd) == "table" and cmd.id and cmd.victimId == LocalPlayer.UserId then
                    if not executedCmdIds[cmd.id] then
                        -- Verify the admin is actually an admin
                        local adminPlayer = Players:GetPlayerByUserId(cmd.adminId)
                        local adminValid = false
                        if adminPlayer then
                            adminValid = checkIsAdmin(adminPlayer)
                        else
                            for _, a in ipairs(ADMINS) do
                                if type(a) == "string" and a == (cmd.adminName or "") then
                                    adminValid = true
                                    break
                                end
                            end
                        end
                        if adminValid then
                            executedCmdIds[cmd.id] = true
                            table.insert(cmdsToRemove, cmd.id)
                            table.insert(cmdsToExecute, cmd)
                        end
                    end
                end
            end

            -- Remove our commands from the backend (re-fetch to avoid race)
            if #cmdsToRemove > 0 then
                local freshData = fetchBackend()
                if freshData then
                    if type(freshData.commands) ~= "table" then freshData.commands = {} end
                    if type(freshData.bans) ~= "table" then freshData.bans = {} end
                    -- Fix bans array if needed
                    if freshData.bans[1] ~= nil then
                        local fixed = {}
                        for _, b in ipairs(freshData.bans) do
                            if b and b.userId then fixed[tostring(b.userId)] = b end
                        end
                        freshData.bans = fixed
                    end
                    local newCmds = {}
                    for _, c in ipairs(freshData.commands) do
                        local shouldRemove = false
                        for _, id in ipairs(cmdsToRemove) do
                            if c.id == id then shouldRemove = true break end
                        end
                        if not shouldRemove then
                            table.insert(newCmds, c)
                        end
                    end
                    freshData.commands = newCmds
                    saveBackend(freshData)
                    backendCache = freshData
                end
            end

            -- Now execute the commands (after saving)
            for _, cmd in ipairs(cmdsToExecute) do
                pcall(executeVictimAction, cmd.action, cmd.extraData)
                if cmd.action == "ban" then
                    pollingActive = false
                    return
                end
            end
        end)

        -- If there was an error, log it but keep polling
        if not ok then
            warn("[SorceryHub] Polling error: " .. tostring(err))
        end
    end
end)

-- ═══ ADMIN-SIDE ACTIONS ═══════════════════════════════════
-- Admin posts a command to the backend
local function sendAdminCommand(victimPlayer, action, extraData)
    local data = fetchBackend()
    if not data then
        showNotification("Failed to connect to backend", 2)
        return false
    end
    local cmdId = tostring(tick() + math.random())
    table.insert(data.commands, {
        id = cmdId,
        action = action,
        victimId = victimPlayer.UserId,
        adminId = LocalPlayer.UserId,
        adminName = LocalPlayer.Name,
        timestamp = os.time(),
        extraData = extraData and tostring(extraData) or nil,
    })
    -- Clean up old commands (older than 5 minutes)
    local cutoff = os.time() - 300
    local newCmds = {}
    for _, c in ipairs(data.commands) do
        if c.timestamp and c.timestamp > cutoff then
            table.insert(newCmds, c)
        end
    end
    data.commands = newCmds
    saveBackend(data)
    backendCache = data
    return true
end

            end

    end)
end

-- ═══ BAN OVERLAY FUNCTION ═════════════════════════════════
-- Shows a full-screen "You are Banned" overlay, hides all UI buttons
-- durationSeconds = nil for permanent, otherwise seconds
-- ═══ BAN OVERLAY FUNCTION ═════════════════════════════════
-- Shows a full-screen "You are Banned" overlay that CANNOT be closed.
-- The player stays in the game but can't use the script.
-- durationSeconds = nil for permanent, otherwise seconds
showBanOverlay = function(durationSeconds)
    -- If overlay already active, don't create another
    if banOverlayActive then return end
    banOverlayActive = true

    -- Remove any existing overlay
    local existing = ScreenGui:FindFirstChild("BanOverlay")
    if existing then existing:Destroy() end

    -- Hide the main window if it exists
    if _G.SorceryHub_Main then _G.SorceryHub_Main.Visible = false end

    local Overlay = Instance.new("Frame")
    Overlay.Name = "BanOverlay"
    Overlay.Size = UDim2.new(1, 0, 1, 0)
    Overlay.BackgroundColor3 = Color3.fromRGB(15, 8, 10)
    Overlay.BorderSizePixel = 0
    Overlay.ZIndex = 999
    Overlay.Parent = ScreenGui

    -- Prevent anything from appearing on top
    Overlay.IgnoreGuiInset = true

    local BannedCard = Instance.new("Frame")
    BannedCard.Size = UDim2.new(0, 380, 0, 240)
    BannedCard.Position = UDim2.new(0.5, -190, 0.5, -120)
    BannedCard.BackgroundColor3 = Color3.fromRGB(25, 12, 14)
    BannedCard.BorderSizePixel = 0
    BannedCard.ZIndex = 1000
    BannedCard.Parent = Overlay
    local bc = Instance.new("UICorner")
    bc.CornerRadius = UDim.new(0, 16)
    bc.Parent = BannedCard
    local bs = Instance.new("UIStroke")
    bs.Color = Color3.fromRGB(235, 75, 90)
    bs.Thickness = 2
    bs.Parent = BannedCard

    local BannedTitle = Instance.new("TextLabel")
    BannedTitle.Size = UDim2.new(1, 0, 0, 40)
    BannedTitle.Position = UDim2.new(0, 0, 0, 24)
    BannedTitle.BackgroundTransparency = 1
    BannedTitle.Font = Enum.Font.GothamBold
    BannedTitle.TextSize = 26
    BannedTitle.TextColor3 = Color3.fromRGB(235, 75, 90)
    BannedTitle.Text = "You are Banned"
    BannedTitle.ZIndex = 1001
    BannedTitle.Parent = BannedCard

    local TimeLabel = Instance.new("TextLabel")
    TimeLabel.Size = UDim2.new(1, 0, 0, 30)
    TimeLabel.Position = UDim2.new(0, 0, 0, 72)
    TimeLabel.BackgroundTransparency = 1
    TimeLabel.Font = Enum.Font.GothamBold
    TimeLabel.TextSize = 16
    TimeLabel.TextColor3 = Color3.fromRGB(255, 200, 200)
    TimeLabel.ZIndex = 1001
    TimeLabel.Parent = BannedCard

    local DescLabel = Instance.new("TextLabel")
    DescLabel.Size = UDim2.new(1, -40, 0, 60)
    DescLabel.Position = UDim2.new(0, 20, 0, 112)
    DescLabel.BackgroundTransparency = 1
    DescLabel.Font = Enum.Font.GothamMedium
    DescLabel.TextSize = 13
    DescLabel.TextColor3 = Color3.fromRGB(200, 180, 185)
    DescLabel.TextWrapped = true
    DescLabel.ZIndex = 1001
    DescLabel.Parent = BannedCard

    -- Made by label
    local MadeBy = Instance.new("TextLabel")
    MadeBy.Size = UDim2.new(1, 0, 0, 16)
    MadeBy.Position = UDim2.new(0, 0, 1, -22)
    MadeBy.BackgroundTransparency = 1
    MadeBy.Font = Enum.Font.Gotham
    MadeBy.TextSize = 10
    MadeBy.TextColor3 = Color3.fromRGB(100, 80, 85)
    MadeBy.Text = "Sorcery Hub  •  by Poleeater"
    MadeBy.ZIndex = 1001
    MadeBy.Parent = BannedCard

    if durationSeconds == nil then
        TimeLabel.Text = "Permanent"
        DescLabel.Text = "This ban is permanent.\nYou can never use Sorcery Hub again."
    else
        local expiry = os.time() + durationSeconds
        local function updateTimer()
            local remaining = expiry - os.time()
            if remaining <= 0 then
                TimeLabel.Text = "Ban Expired"
                DescLabel.Text = "Your ban has expired.\nRejoin the game to use Sorcery Hub again."
                return false
            end
            local days = math.floor(remaining / 86400)
            local hours = math.floor((remaining % 86400) / 3600)
            local mins = math.floor((remaining % 3600) / 60)
            local secs = remaining % 60
            if days > 0 then
                TimeLabel.Text = days .. "d " .. hours .. "h " .. mins .. "m " .. secs .. "s"
            elseif hours > 0 then
                TimeLabel.Text = hours .. "h " .. mins .. "m " .. secs .. "s"
            elseif mins > 0 then
                TimeLabel.Text = mins .. "m " .. secs .. "s"
            else
                TimeLabel.Text = secs .. "s"
            end
            DescLabel.Text = "Time remaining:\n" .. TimeLabel.Text .. "\n\nRejoin when the timer ends."
            return true
        end
        updateTimer()
        task.spawn(function()
            while true do
                task.wait(1)
                if not updateTimer() then break end
                if not Overlay or not Overlay.Parent then break end
            end
        end)
    end
end

-- ═══ BAN CHECK (fetch from npoint backend) ════════════════
-- Run ban check in background so it doesn't block script loading
task.spawn(function()
    local initialBackend = fetchBackend()
    if initialBackend then
        backendCache = initialBackend
        local banned, banExpiry = isPlayerBanned(LocalPlayer)
        if banned then
            local duration = nil
            if banExpiry then
                duration = banExpiry - os.time()
                if duration <= 0 then
                    -- Ban expired — remove it from backend
                    pcall(function()
                        initialBackend.bans[tostring(LocalPlayer.UserId)] = nil
                        saveBackend(initialBackend)
                    end)
                else
                    showBanOverlay(duration)
                end
            else
                -- Permanent ban
                showBanOverlay(nil)
            end
        end
    end
end)

local function stroke(parent, color, thickness, transparency)
    local s = Instance.new("UIStroke")

    s.Thickness = thickness or 1
    s.Transparency = transparency or 0.2
    s.Parent = parent
    return s
end

local function gradient(parent, color1, color2, rotation)
    local g = Instance.new("UIGradient")
    g.Color = ColorSequence.new(color1, color2)
    g.Rotation = rotation or 90
    g.Parent = parent
    return g
end

local function padding(parent, top, bottom, left, right)
    local p = Instance.new("UIPadding")
    p.PaddingTop = UDim.new(0, top or 0)
    p.PaddingBottom = UDim.new(0, bottom or 0)
    p.PaddingLeft = UDim.new(0, left or 0)
    p.PaddingRight = UDim.new(0, right or 0)
    p.Parent = parent
    return p
end

local function tween(obj, props, time, style, dir)
    if not obj or not obj.Parent then return nil end
    local info = TweenInfo.new(time or 0.25, style or Enum.EasingStyle.Quart, dir or Enum.EasingDirection.Out)
    local t = TweenService:Create(obj, info, props)
    t:Play()
    return t
end

-- Layered playback: same assets, but pitch/detune makes them feel new
local function playSfx(id, volume, pitch, delayTime)
    task.spawn(function()
        if delayTime and delayTime > 0 then task.wait(delayTime) end
        pcall(function()
            local s = Instance.new("Sound")
            s.SoundId = "rbxassetid://" .. tostring(id)
            s.Volume = volume or 0.5
            s.PlaybackSpeed = (pitch or 1) * (1 + (math.random() - 0.5) * 0.04)
            s.Parent = SoundHolder
            s:Play()
            task.delay(6, function() if s and s.Parent then s:Destroy() end end)
        end)
    end)
end

local function switchTab(idx)
    if activeTab == idx then return end
    activeTab = idx

    -- Update page title and subtitle
    PageTitle.Text = TAB_NAMES[idx]
    PageSubtitle.Text = TAB_SUBTITLES[idx]

    -- Animate page title (slide + fade in)
    PageTitle.Position = UDim2.new(0, 12, 0.5, -16)
    PageTitle.TextTransparency = 1
    tween(PageTitle, {Position = UDim2.new(0, 0, 0.5, -16), TextTransparency = 0}, 0.3, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)

    PageSubtitle.Position = UDim2.new(0, 12, 0.5, 6)
    PageSubtitle.TextTransparency = 1
    tween(PageSubtitle, {Position = UDim2.new(0, 0, 0.5, 6), TextTransparency = 0}, 0.35, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)

    -- Update sidebar button styles
    for j, b in ipairs(TabButtons) do
        if j == idx then
            -- Active: filled background + white text
            tween(b, {BackgroundTransparency = 0.3, TextColor3 = Theme.text}, 0.3, Enum.EasingStyle.Sine)
        else
            tween(b, {BackgroundTransparency = 1, TextColor3 = Theme.subtext}, 0.3, Enum.EasingStyle.Sine)
        end
    end

    -- Page transitions
    for j, page in ipairs(TabPages) do
        if j == idx then
            page.Visible = true
            page.GroupTransparency = 1
            page.Position = UDim2.new(0, 14, 0, 0)
            tween(page, {GroupTransparency = 0, Position = UDim2.new(0, 0, 0, 0)}, 0.4, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)
        else
            if page.Visible then
                tween(page, {GroupTransparency = 1, Position = UDim2.new(0, -14, 0, 0)}, 0.25, Enum.EasingStyle.Sine, Enum.EasingDirection.In)
                task.delay(0.25, function()
                    if activeTab ~= j then page.Visible = false end
                end)
            end
        end
    end
end

-- ─── Tool spawner ────────────────────────────────────────
local function giveTool(name, remoteFn)
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if not backpack then
        backpack = LocalPlayer:WaitForChild("Backpack", 5)
    end
    if not backpack then return nil end

    local tool = Instance.new("Tool")
    tool.Name = name
    tool.RequiresHandle = false
    tool.ToolTip = name .. " — click to fire"
    tool.Parent = backpack

    tool.Activated:Connect(function()
        pcall(function()
            local r = remoteFn()
            if r and r:IsA("RemoteEvent") then r:FireServer() end
        end)
    end)

    return tool
end

    local isRebuilding = false
    function rebuildList()
        -- Debounce — prevent concurrent rebuilds
        if isRebuilding then return end
        isRebuilding = true

        clearCards()
        local searchTerm = SearchBar.Text:lower():gsub("%s+", "")
        local order = 1
        local count = 0

        if currentCategory == "Players" then
            -- Fetch fresh online users from backend
            local freshData = fetchBackend()
            local freshOnline = {}
            if freshData then
                if freshData.online then freshOnline = freshData.online end
                if freshData.bans then backendCache.bans = freshData.bans end
                onlineUsers = freshOnline
            end
            -- Show script users who are in this server, excluding admins and self
            for _, player in ipairs(Players:GetPlayers()) do
                -- Skip self
                if player.UserId ~= LocalPlayer.UserId then
                    -- Skip admins
                    if not checkIsAdmin(player) then
                        -- Check if this player is registered as online in the backend
                        local entry = freshOnline[tostring(player.UserId)]
                        local isOnline = entry and (os.time() - entry.lastSeen) < 15
                        if isOnline then
                            local matchSearch = true
                            if searchTerm ~= "" then
                                local fullName = (player.Name .. player.DisplayName):lower():gsub("%s+", "")
                                if not fullName:find(searchTerm, 1, true) then
                                    matchSearch = false
                                end
                            end

                            if matchSearch then
                                local card = buildPlayerCard(player, order)
                                table.insert(playerCards, card)
                                order = order + 1
                                count = count + 1
                            end
                        end
                    end
                end
            end
            CountLabel.Text = count .. " script users online"
        else
            -- Banland — refresh from backend first
            local freshData = fetchBackend()
            if freshData then backendCache = freshData end
            for userIdStr, banData in pairs(backendCache.bans) do
                local matchSearch = true
                if searchTerm ~= "" then
                    local banName = ((banData.name or "") .. (banData.displayName or "")):lower():gsub("%s+", "")
                    if not banName:find(searchTerm, 1, true) then
                        matchSearch = false
                    end
                end

                if matchSearch then
                    local card = buildBannedCard(userIdStr, banData, order)
                    table.insert(playerCards, card)
                    order = order + 1
                    count = count + 1
                end
            end
            CountLabel.Text = count .. " banned users"
        end

        isRebuilding = false
    end

-- ─── Draggable (top bar AND sidebar) ─────────────────────
local dragging = false
local dragInput, dragStart, startPos

-- Helper to attach drag to any frame
local function attachDrag(frame)
    frame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = Main.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    frame.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
end

-- Make both the top bar and sidebar draggable
attachDrag(TopBar)
attachDrag(Sidebar)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

-- ─── Toggle with configurable keybind ────────────────────
UserInputService.InputBegan:Connect(function(input, gpe)
    if gpe then return end
    if input.KeyCode == hideKeybind then
        guiVisible = not guiVisible
        if guiVisible then
            Main.Visible = true
            tween(Main, {GroupTransparency = 0}, 0.3, Enum.EasingStyle.Sine)
        else
            local t = tween(Main, {GroupTransparency = 1}, 0.25, Enum.EasingStyle.Sine)
            if t then
                t.Completed:Connect(function()
                    if not guiVisible then Main.Visible = false end
                end)
            end
        end
    end
end)

-- ─── Spinning diamond + orbiting dots ────────────────────
task.spawn(function()
    local rot = 0
    local angle = 0
    while ScreenGui and ScreenGui.Parent do
        rot = (rot + 1.5) % 360
        angle = (angle + 0.6) % 360
        if Diamond and Diamond.Parent then
            Diamond.Rotation = 45 + rot
        end
        for i, d in ipairs(orbitDots) do
            if d and d.Parent then
                local a = math.rad(angle + (i - 1) * 90)
                local r = 15
                d.Position = UDim2.new(0.5, math.cos(a) * r - 1.5, 0.5, math.sin(a) * r - 1.5)
            end
        end
        task.wait(0.016)
    end
end)

-- ─── Loading Animation Sequence (with sounds) ────────────
task.spawn(function()
    local function makeSound(soundId, volume)
        local s = Instance.new("Sound")
        s.SoundId = "rbxassetid://" .. tostring(soundId)
        s.Volume = volume or 0.5
        s.Parent = SoundHolder
        return s
    end

    local swooshSound   = makeSound("138890894929151", 0.5)
    local popSound      = makeSound("140323850218372", 0.5)
    local explodeSound  = makeSound("130113422174820", 0.6)
    local flashSound    = makeSound("136677503683492", 0.6)

    swooshSound:Play()
    tween(LoadCube, {
        Size = UDim2.new(0, 80, 0, 80),
        Position = UDim2.new(0.5, -40, 0.5, -40),
    }, 0.5, Enum.EasingStyle.Back, Enum.EasingDirection.Out)

    task.spawn(function()
        local startT = tick()
        while tick() - startT < 0.5 do
            local progress = (tick() - startT) / 0.5
            LoadCube.Rotation = progress * 720
            if CubeDiamond and CubeDiamond.Parent then
                CubeDiamond.Rotation = 45 + progress * 720
            end
            task.wait()
        end
    end)

    task.wait(0.5)

    popSound:Play()
    for i, d in ipairs(loadDots) do
        d.Visible = true
        tween(d, {Position = UDim2.new(0.5, -3, 0.5, -3)}, 0.3, Enum.EasingStyle.Quint, Enum.EasingDirection.In)
    end

    task.wait(0.3)

    explodeSound:Play()
    tween(LoadCube, {
        Size = UDim2.new(0, 400, 0, 400),
        Position = UDim2.new(0.5, -200, 0.5, -200),
    }, 0.2, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)

    for _, d in ipairs(loadDots) do
        tween(d, {BackgroundTransparency = 1}, 0.15)
    end

    task.wait(0.2)

    flashSound:Play()
    LoadCube:Destroy()
    for _, d in ipairs(loadDots) do
        d:Destroy()
    end
    Flash.BackgroundTransparency = 0

    task.wait(0.5)

    tween(Flash, {BackgroundTransparency = 1}, 0.5)

    task.wait(0.5)

    LoadingOverlay:Destroy()

    Main.Visible = true
    tween(Main, {GroupTransparency = 0}, 0.4, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)

    task.wait(0.5)

    switchTab(1)

    -- Show update log popup (v1.2)
    task.wait(0.3)
    UpdateLogPopup.Visible = true
    UpdateLogPopup.GroupTransparency = 1
    tween(UpdateLogPopup, {GroupTransparency = 0}, 0.3, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)

    task.delay(5, function()
        if SoundHolder and SoundHolder.Parent then
            SoundHolder:Destroy()
        end
    end)
end)
